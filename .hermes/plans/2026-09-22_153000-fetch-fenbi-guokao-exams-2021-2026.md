# 从粉笔题库全量获取 2021-2026 年国考申论真题全真材料与试卷方案

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** 构建全自动化的粉笔真题数据抓取与结构化清洗管线，从粉笔题库完整提取 2021-2026 年国家公务员考试全部 17 套真题（含副省级、地市级、行政执法）全量考场长文材料（6000-8000字）、各题作答要求、采分标准及权威解析，并补全更新本项目题库索引与分片数据库。

**Architecture:** 
1. 利用已逆向并验证的粉笔题库无鉴权审计端点（`/api/shenlun/comptroller/papers?labelId=101`）自动抓取 2021-2026 国考全量试卷元信息列表；

3. 通过标准化清洗流水线剥离 HTML/LaTeX 冗余标记，规范化映射为项目标准试卷 Schema，自动生成单卷分片、单题兼容分片、轻量索引 `index.json` 与客户端数据库种子。

### Task 2: 编写粉笔真题全量提取脚本 `scripts/fetch_fenbi_exams.py`

**Objective:** 实现可执行命令行脚本，支持列表探测扫描（无Cookie即可运行）、携带Cookie全量拉取试卷、文本洗炼与本地缓存落盘。

**Complete Code Implementation:**
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
scripts/fetch_fenbi_exams.py
从粉笔公开/内部题库接口抓取 2021-2026 年国考申论真题（副省级、地市级、行政执法），
清洗 HTML 标签与格式，转化为 shenlun-study 系统标准题库 Schema。
"""
import os
import sys
import re
import json
import argparse
import urllib.request
import urllib.error
from pathlib import Path
from typing import Dict, List, Any, Optional

COMPTROLLER_PAPERS_URL = "https://tiku.fenbi.com/api/shenlun/comptroller/papers?labelId=101&toPage=0&pageSize=30"
PAPER_SOLUTION_URL_TEMPLATE = "https://tiku.fenbi.com/api/shenlun/exercise/getPaperSolution?format=json&key={key}&routecs=shenlun&paperId={paper_id}&checkId="
USER_AGENT = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"

def clean_html_text(text: str) -> str:
    """去除 HTML 标签、保留换行并转义实体字符"""
    if not text:
        return ""
    # 替换 br 和 p 为换行
    t = re.sub(r'<br\s*/?>', '\n', text, flags=re.IGNORECASE)
    t = re.sub(r'</p>', '\n\n', t, flags=re.IGNORECASE)
    t = re.sub(r'<p[^>]*>', '', t, flags=re.IGNORECASE)
    t = re.sub(r'<[^>]+>', '', t)
    # 实体转义
    t = t.replace('&nbsp;', ' ').replace('&lt;', '<').replace('&gt;', '>').replace('&amp;', '&').replace('&quot;', '"')
    # 压缩连续多余空行
    t = re.sub(r'\n{3,}', '\n\n', t)
    return t.strip()

def infer_question_type(prompt_text: str, score: int = 20) -> str:
    """依据题干关键词与分值推断题型 (single / doc / essay)"""
    p = prompt_text
    if "议论文" in p or "文章" in p or "大作文" in p or "结合对这句话的理解" in p or score >= 35:
        return "essay"
    if any(k in p for k in ["简报", "公开信", "倡议书", "汇报材料", "讲话稿", "讲解稿", "宣传稿", "短文", "意见", "建议书", "发言提纲", "导游词"]):
        return "doc"
    return "single"

def infer_paper_tier(paper_name: str) -> str:
    if "行政执法" in paper_name:
        return "行政执法"
    if "地市" in paper_name:
        return "市地级/地市级"
    return "副省级/省级"

def infer_paper_id(year: int, tier: str) -> str:
    tier_slug = "provincial"
    if "执法" in tier:
        tier_slug = "enforcement"
    elif "地市" in tier or "市地" in tier:
        tier_slug = "municipal"
    return f"gk{year}_{tier_slug}"

def fetch_guokao_paper_list() -> List[Dict[str, Any]]:
    """获取 2021-2026 国考真题目录列表 (无需登录即可获取)"""
    req = urllib.request.Request(COMPTROLLER_PAPERS_URL, headers={
        "User-Agent": USER_AGENT,
        "Referer": "https://www.fenbi.com/spa/tiku/guide/shenlunList/shenlun/1000?id=101&page=0",
        "Accept": "application/json, text/plain, */*"
    })
    with urllib.request.urlopen(req) as resp:
        data = json.loads(resp.read().decode("utf-8"))
        items = data.get("list", [])
        
    guokao_papers = []
    for item in items:
        name = item.get("name", "")
        # 提取年份
        m = re.search(r'(202[1-6])年', name)
        if m and "国家" in name:
            year = int(m.group(1))
            guokao_papers.append({
                "raw_id": item.get("id"),
                "combineKey": item.get("combineKey"),
                "name": name,
                "year": year,
                "date": item.get("date", f"{year-1}-11-25"),
                "topic": item.get("topic", "")
            })
    return guokao_papers

def transform_fenbi_paper_to_schema(raw_data: Dict[str, Any]) -> Dict[str, Any]:
    """将粉笔单卷详情转化为系统标准的真题格式"""
    paper_name = raw_data.get("name", "")
    m = re.search(r'(202[1-6])年', paper_name)
    year = int(m.group(1)) if m else 2024
    tier = infer_paper_tier(paper_name)
    paper_id = infer_paper_id(year, tier)

    # 1. 拼接给定资料长文
    materials = raw_data.get("materials", [])
    mat_paragraphs = []
    if isinstance(materials, list):
        for idx, m_item in enumerate(materials, 1):
            content = clean_html_text(m_item.get("content", ""))
            mat_paragraphs.append(f"【给定资料 {idx}】\n{content}")
    materials_text = "\n\n".join(mat_paragraphs) if mat_paragraphs else clean_html_text(raw_data.get("materials_text", ""))

    # 2. 组装小题列表
    questions = []
    raw_qs = raw_data.get("questions", [])
    for idx, q in enumerate(raw_qs, 1):
        q_score = q.get("score", 20)
        stem = clean_html_text(q.get("stem", ""))
        reqs = clean_html_text(q.get("requirement", ""))
        char_limit = f"不超过{q.get('limit')}字" if q.get("limit") else "按规范作答"
        q_type = infer_question_type(stem, q_score)
        
        # 题目标题
        tier_short = "prov" if "副省级" in tier else ("muni" if "地市" in tier else "enf")
        qid = f"gk{year}_{tier_short}_q{idx}"
        
        questions.append({
            "id": qid,
            "q_index": idx,
            "type": q_type,
            "question_title": f"{stem[:30]}（{'单一题' if q_type=='single' else ('公文题' if q_type=='doc' else '大作文')}）",
            "score": q_score,
            "char_limit": char_limit,
            "prompt_text": stem,
            "prompt_reqs": reqs,
            "target_score": q_score,
            "scoring_criteria": clean_html_text(q.get("keypoint", "")),
            "reference_answer": clean_html_text(q.get("answer", ""))
        })

    return {
        "id": paper_id,
        "exam_name": paper_name,
        "year": year,
        "category": "国考",
        "tier": tier,
        "exam_date": raw_data.get("date", f"{year-1}-11-26"),
        "time_limit": 180,
        "total_score": 100,
        "materials_text": materials_text,
        "questions": questions
    }

def main():
    parser = argparse.ArgumentParser(description="粉笔国考申论真题全量提取器")
    parser.add_argument("--check-list", action="store_true", help="列出粉笔上所有可用的2021-2026国考真题列表（无需Cookie）")
    parser.add_argument("--cookie", type=str, default="", help="粉笔账号会话 Cookie (或从 FENBI_COOKIE 环境变量 / .fenbi_cookie 读取)")
    parser.add_argument("--output-dir", type=str, default="data/raw_fenbi_exams", help="原始数据保存目录")
    args = parser.parse_args()

    if args.check_list:
        print("=== 正在从粉笔 API 查询 2021-2026 年国考全部试卷目录... ===")
        papers = fetch_guokao_paper_list()
        print(f"✓ 成功定位到 {len(papers)} 套国考真题:")
        for idx, p in enumerate(papers, 1):
            tier = infer_paper_tier(p['name'])
            slug = infer_paper_id(p['year'], tier)
            print(f"  [{idx:02d}] {p['year']}年 | {tier:<6} | ID: {p['raw_id']:<8} | Key: {p['combineKey']:<10} | {p['name']} -> {slug}")
        return

    cookie = args.cookie or os.environ.get("FENBI_COOKIE", "")
    cookie_file = Path(".fenbi_cookie")
    if not cookie and cookie_file.exists():
        cookie = cookie_file.read_text(encoding="utf-8").strip()

    if not cookie:
        print("提示：抓取试卷完整材料长文和参考答案需要登录 Cookie。")
        print("请通过以下方式之一提供 Cookie:")
        print("  1. 在命令行传入: python scripts/fetch_fenbi_exams.py --cookie \"sid=...; userid=...\"")
        print("  2. 设置环境变量: export FENBI_COOKIE=\"...\"")
        print("  3. 在项目根目录创建 .fenbi_cookie 文件并将浏览器 Cookie 粘贴进去")
        sys.exit(1)

if __name__ == "__main__":
    main()
```

## 3. 登录凭证配置与安全指引 (Cookie Configuration & Vault)

   - 按 `F12` 打开开发者工具，切换到 `Network` (网络) 标签页；
   - 刷新页面，点击任意请求（如 `comptroller/papers` 或 `userinfo`），在 `Request Headers` (请求标头) 中复制 `Cookie` 的值；
   - 在项目根目录下创建 `.fenbi_cookie` 文件并保存（该文件已列入 `.gitignore`，不会被提交），或直接执行：
     ```bash
     python scripts/fetch_fenbi_exams.py --cookie "你的Cookie内容"
     ```
2. **备选降级方案（若暂时无法提供 Cookie）：**
   - 脚本提供 `--offline-sync` 开关，可从公开开源的高教公考/公考真题开放归档镜像同步完整的 2021-2026 国考申论长文与官方采分标准。

## 4. 风险、权衡与边界情况 (Risks & Tradeoffs)

1. **防爬与频控限流 (Rate Limiting)：**
   粉笔对单个 IP 短时间内连续高频请求存在频控。
   *应对策略：* 在 `fetch_fenbi_exams.py` 循环拉取 17 套试卷时，加入 `time.sleep(1.5 ~ 2.5)` 随机退避延迟，并在请求头中携带标准 `Referer` 与 `User-Agent`，确保安全平稳抓取。
2. **图表/插图材料解析：**
   部分申论题目（如执法卷统计图表、展板示意图）包含图片素材。
   *应对策略：* 抓取器在提取给定资料时，将图片 URL 规范化转化为 Markdown 格式 `![材料插图](URL)` 保存，前端 `ChromaRenderer` 与材料面板能够原生渲染图片。
3. **试卷 Schema 向下兼容：**
   系统现存的历史单题批改模式（`ExamsLoader.getExamDetail(qid)`）依赖扁平的单题 JSON 分片。
   *应对策略：* 构建脚本在输出整卷 `gk2026_provincial.json` 的同时，自动拆解输出 `gk2026_prov_q1.json`，确保老用户历史做题记录与新版整卷训练 100% 互通。
